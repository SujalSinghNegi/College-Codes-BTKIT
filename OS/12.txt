import math

def calculate_distance(p1, p2):
    return math.sqrt((p1[0] - p2[0])**2 + (p1[1] - p2[1])**2)

def tsp_nearest_neighbor(cities):
    unvisited = list(cities.keys())
    start_city = unvisited[0]
    current_city = start_city
    
    unvisited.remove(current_city)
    path = [current_city]
    total_distance = 0
    
    while unvisited:
        nearest_city = min(
            unvisited, 
            key=lambda city: calculate_distance(cities[current_city], cities[city])
        )
        
        total_distance += calculate_distance(cities[current_city], cities[nearest_city])
        current_city = nearest_city
        unvisited.remove(current_city)
        path.append(current_city)
    
    total_distance += calculate_distance(cities[current_city], cities[start_city])
    path.append(start_city)
    
    return path, total_distance

cities = {
    'A': (0, 0),
    'B': (1, 5),
    'C': (4, 3),
    'D': (2, 8),
    'E': (6, 1)
}

optimal_path, min_distance = tsp_nearest_neighbor(cities)

print("Route:", " -> ".join(optimal_path))
print("Distance:", round(min_distance, 2))

import nltk
from nltk.corpus import stopwords
from nltk.tokenize import word_tokenize

nltk.download('punkt')
nltk.download('stopwords')

def remove_stopwords(text):
    stop_words = set(stopwords.words('english'))
    words = word_tokenize(text)
    filtered_words = [word for word in words if word.lower() not in stop_words]
    filtered_text = " ".join(filtered_words)
    return filtered_text

if __name__ == "__main__":
    sample_passage = "This is a simple example passage to demonstrate how we can easily remove stop words using the NLTK library in Python."
    
    print("Original Passage:")
    print(sample_passage)
    print("-" * 40)
    
    result = remove_stopwords(sample_passage)
    
    print("Filtered Passage:")
    print(result)
