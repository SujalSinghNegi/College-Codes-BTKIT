import math

def calc(p1, p2):
    return math.sqrt((p1[0] - p2[0])**2 + (p1[1] - p2[1])**2)

def tsp(cities):
    unvisited = list(cities.keys())
    start = unvisited[0]
    cur = start
    
    unvisited.remove(cur)
    path = [cur]
    dist = 0
    
    while unvisited:
        near = min(
            unvisited, 
            key=lambda city: calc(cities[cur], cities[city])
        )
        
        dist += calc(cities[cur], cities[near])
        cur = near
        unvisited.remove(cur)
        path.append(cur)
    
    dist += calc(cities[cur], cities[start])
    path.append(start)
    
    return path, dist

cities = {
    'A': (0, 0),
    'B': (1, 5),
    'C': (4, 3),
    'D': (2, 8),
    'E': (6, 1)
}

optimal_path, minDist = tsp(cities)

print("Route:", " -> ".join(optimal_path))
print("Distance:", round(minDist, 2))

import nltk
from nltk.corpus import stopwords
from nltk.tokenize import word_tokenize

nltk.download('punkt')
nltk.download('stopwords')

with open("input.txt", "r") as file: 
    text = file.read()

words = word_tokenize(text)

stop_words = set(stopwords.words('english'))

filtered = [w for w in words if w.lower() not in stop_words]

print("Filtered Text: ", " ".join(filtered))
